// ============================================================
//  설정 부분 (여기만 수정하면 됩니다)
// ============================================================

// 테스트 모드: true 이면 모든 메일이 TEST_EMAIL 주소로만 발송됩니다.
// 실제 고객에게 보낼 준비가 되면 false 로 바꾸세요.
const TEST_MODE = true;

// 테스트 모드일 때 메일을 받을 내 주소 (비워두면 현재 로그인한 계정으로 자동 설정)
const TEST_EMAIL = '';

// 작업할 시트 이름
const SHEET_NAME = '고객명단';

// 메일 제목
const MAIL_SUBJECT = '[안내] 휴면 계정 안내 메일입니다';


// ============================================================
//  메인 함수: 이 함수를 실행하세요
// ============================================================

// C열이 '휴면'이고 G열이 비어 있는 고객에게 메일을 보내고, G열에 오늘 날짜를 기록하는 함수
function sendDormantEmails() {

  // 현재 열려 있는 스프레드시트 파일을 가져옵니다.
  const spreadsheet = SpreadsheetApp.getActiveSpreadsheet();

  // 그 파일 안에서 '고객명단' 시트를 찾습니다.
  const sheet = spreadsheet.getSheetByName(SHEET_NAME);

  // 시트를 못 찾았으면 (null이면) 안내 메시지를 남기고 종료합니다.
  if (sheet === null) {
    // 실행 로그에 메시지를 남깁니다.
    Logger.log("'" + SHEET_NAME + "' 시트를 찾을 수 없습니다. 시트 이름을 확인해 주세요.");
    // 함수를 여기서 끝냅니다.
    return;
  }

  // 테스트 메일을 받을 주소를 담을 변수입니다. (값이 바뀔 수 있으므로 let)
  let myEmail = TEST_EMAIL;

  // 테스트 모드인데 주소가 비어 있으면
  if (TEST_MODE && myEmail === '') {
    // 현재 로그인한 구글 계정 주소를 사용합니다.
    myEmail = Session.getActiveUser().getEmail();
  }

  // 데이터가 들어 있는 마지막 행 번호를 구합니다.
  const lastRow = sheet.getLastRow();

  // 마지막 행이 2보다 작으면 (제목행만 있으면) 처리할 데이터가 없습니다.
  if (lastRow < 2) {
    // 안내 메시지를 남깁니다.
    Logger.log('처리할 데이터가 없습니다.');
    // 함수를 끝냅니다.
    return;
  }

  // 2행부터 마지막 행까지, A열(1)부터 7개 열(A~G)을 한 번에 읽어옵니다.
  // getRange(시작행, 시작열, 행 개수, 열 개수)
  const data = sheet.getRange(2, 1, lastRow - 1, 7).getValues();

  // 오늘 날짜를 '2025-01-01' 형식의 글자로 만듭니다. (한국 시간 기준)
  const today = Utilities.formatDate(new Date(), 'Asia/Seoul', 'yyyy-MM-dd');

  // 보낸 메일 개수를 세는 변수입니다. (1씩 늘어나므로 let)
  let sentCount = 0;

  // 읽어온 데이터를 첫 줄부터 마지막 줄까지 한 줄씩 반복합니다.
  for (let i = 0; i < data.length; i++) {

    // A열 값: 이름 (배열은 0부터 세므로 A열은 0번)
    const name = data[i][0];

    // B열 값: 이메일 (1번)
    const email = data[i][1];

    // C열 값: 등급 (2번)
    const grade = data[i][2];

    // G열 값: 발송여부 (6번)
    const sent = data[i][6];

    // 실제 시트에서의 행 번호 (데이터가 2행부터 시작하므로 i + 2)
    const rowNumber = i + 2;

    // 등급이 '휴면'이 아니거나, 이미 발송 기록이 있으면 이 행은 건너뜁니다.
    if (grade !== '휴면' || sent !== '') {
      // 다음 행으로 넘어갑니다.
      continue;
    }

    // 이메일 주소가 비어 있으면 보낼 수 없으므로 건너뜁니다.
    if (email === '') {
      // 어느 행이 비어 있었는지 기록을 남깁니다.
      Logger.log(rowNumber + '행: 이메일 주소가 없어서 건너뜁니다.');
      // 다음 행으로 넘어갑니다.
      continue;
    }

    // 메일 본문을 만듭니다. (A열 이름을 맨 앞에 넣습니다. \n 은 줄바꿈)
    const body =
      name + '님, 안녕하세요.\n\n' +
      '회원님의 계정이 휴면 상태로 전환되었습니다.\n' +
      '서비스를 계속 이용하시려면 로그인해 주세요.\n\n' +
      '감사합니다.';

    // 받는 사람 주소: 테스트 모드면 내 주소, 아니면 B열 고객 주소
    // (조건 ? 참일 때 값 : 거짓일 때 값)
    const recipient = TEST_MODE ? myEmail : email;

    // 메일 제목을 담을 변수입니다. (테스트 모드면 내용이 바뀌므로 let)
    let subject = MAIL_SUBJECT;

    // 테스트 모드이면
    if (TEST_MODE) {
      // 제목 앞에 [테스트]를 붙이고, 원래 누구에게 갈 메일인지 표시합니다.
      subject = '[테스트] ' + subject + ' (원래 수신자: ' + email + ')';
    }

    // 메일을 보냅니다. sendEmail(받는사람, 제목, 본문)
    MailApp.sendEmail(recipient, subject, body);

    // 이 행의 G열(7번째 열)에 오늘 날짜를 기록합니다.
    sheet.getRange(rowNumber, 7).setValue(today);

    // 보낸 개수를 1 늘립니다.
    sentCount++;

    // 누구에게 보냈는지 기록을 남깁니다.
    Logger.log(rowNumber + '행: ' + name + ' (' + recipient + ') 에게 발송 완료');
  }

  // 반복이 모두 끝나면 총 발송 개수를 기록합니다.
  Logger.log('총 ' + sentCount + '건 발송 완료 (테스트 모드: ' + TEST_MODE + ')');
}
