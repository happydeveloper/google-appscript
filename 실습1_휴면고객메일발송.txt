/**
 * 실습 1. 휴면 고객 메일 발송
 * ------------------------------------------------------------
 * 시트: '고객명단' (1행 제목행)
 *   A열 이름 / B열 이메일 / C열 등급 / G열 발송여부
 * 기능: C열이 '휴면'이고 아직 발송되지 않은 행에 메일 발송,
 *       발송 결과를 G열에 기록
 * 실행: 편집기에서 sendDormantMails 선택 후 ▶ 직접 실행
 */

// ==============================
// ⚙️ 설정 영역 (여기만 바꿔서 사용하세요)
// ==============================

// 테스트 모드: true면 모든 메일이 아래 MY_EMAIL 주소로만 갑니다 (실제 고객에게 안 감)
//             false로 바꾸면 실제 고객에게 발송됩니다
const TEST_MODE = true;

// 테스트 메일을 받을 내 이메일 주소 (본인 주소로 바꿔서 사용)
const MY_EMAIL = 'your-email@gmail.com';

// 메일 제목과 본문 (본문의 {이름} 자리에 고객 이름이 들어갑니다)
const MAIL_SUBJECT = '오랜만이에요! 다시 찾아주세요';
const MAIL_BODY = '{이름}님, 안녕하세요.\n\n한동안 소식이 없으셔서 안부 인사드립니다.\n다시 만나뵙기를 기다리고 있을게요.\n\n감사합니다.';


// ==============================
// 📧 휴면 고객에게 메일 보내기
// ==============================
function sendDormantMails() {
  // '고객명단' 시트를 가져옵니다
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('고객명단');
  if (!sheet) {
    SpreadsheetApp.getUi().alert("'고객명단' 시트를 찾을 수 없어요.");
    return;
  }

  // 시트의 모든 데이터를 한 번에 읽어옵니다 (표 형태의 배열)
  const data = sheet.getDataRange().getValues();

  // 오늘 날짜를 '2026-10-30' 같은 형식으로 만듭니다
  const today = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd');

  let successCount = 0; // 성공한 메일 수
  let failCount = 0;    // 실패한 메일 수
  let stoppedByQuota = false; // 하루 발송 한도 때문에 멈췄는지 여부

  // 1행은 제목행이므로 2행(i = 1)부터 확인합니다
  for (let i = 1; i < data.length; i++) {
    const row = data[i];

    const name   = String(row[0]).trim(); // A열: 이름
    const email  = String(row[1]).trim(); // B열: 이메일
    const grade  = String(row[2]).trim(); // C열: 등급
    const status = String(row[6]).trim(); // G열: 발송여부

    // 등급이 '휴면'이 아니거나 이메일이 비어 있으면 건너뜁니다
    if (grade !== '휴면' || email === '') {
      continue;
    }

    // G열이 비어 있거나, '테스트'·'실패' 기록만 있는 행은 발송 대상입니다
    // (실제 발송에 성공해서 날짜가 적힌 행만 건너뜁니다)
    const canSend = status === '' || status.startsWith('테스트') || status.startsWith('실패');
    if (!canSend) {
      continue;
    }

    // 오늘 보낼 수 있는 메일이 남아 있는지 확인합니다
    if (MailApp.getRemainingDailyQuota() < 1) {
      stoppedByQuota = true;
      break; // 한도를 다 썼으면 반복을 멈춥니다
    }

    // 본문의 {이름}을 실제 고객 이름으로 바꿉니다
    const body = MAIL_BODY.replace('{이름}', name);

    // G열 칸 위치 (i + 1 = 실제 행 번호, 7 = G열)
    const resultCell = sheet.getRange(i + 1, 7);

    try {
      if (TEST_MODE) {
        // 테스트 모드: 내 주소로 보내고, 원래 받을 사람을 제목에 표시합니다
        MailApp.sendEmail(MY_EMAIL, '[테스트 → ' + email + '] ' + MAIL_SUBJECT, body);
        resultCell.setValue('테스트 발송 ' + today);
      } else {
        // 실제 모드: 고객에게 보내고 오늘 날짜를 기록합니다
        MailApp.sendEmail(email, MAIL_SUBJECT, body);
        resultCell.setValue(today);
      }
      successCount++;
    } catch (error) {
      // 발송 중 오류가 나면 G열에 실패와 이유를 기록합니다
      resultCell.setValue('실패 ' + today + ' : ' + error.message);
      failCount++;
    }
  }

  // 결과를 알림창으로 보여줍니다
  let message = '성공 ' + successCount + '건, 실패 ' + failCount + '건';
  if (TEST_MODE) {
    message += '\n(테스트 모드: ' + MY_EMAIL + '로만 발송)';
  }
  if (stoppedByQuota) {
    message += '\n오늘 발송 한도를 모두 사용해서 중간에 멈췄어요. 내일 다시 실행하면 나머지가 발송됩니다.';
  }
  SpreadsheetApp.getUi().alert(message);
}
